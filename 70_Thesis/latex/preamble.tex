% --- from the faculty template (do not modify) ------------------------
\usepackage{graphicx, color}
\usepackage[utf8]{inputenc}
\usepackage[english]{babel}      % British spelling throughout the prose
\usepackage[a4paper,margin=2cm]{geometry}

% --- maths ------------------------------------------------------------
\usepackage{amsmath, amssymb, amsthm}
\usepackage{bm}
\usepackage{nicefrac}          % \nicefrac{1}{4} in the curvature section

% --- tables and figures ----------------------------------------------
\usepackage{booktabs}            % \toprule \midrule \bottomrule; never \hline
\usepackage{tabularx}
\usepackage{multirow}
\usepackage{subcaption}
\usepackage[font=small,labelfont=bf]{caption}
\usepackage{float}

% --- units and numbers ------------------------------------------------
\usepackage{siunitx}
\DeclareSIUnit{\nothing}{}      % for bare counts: \SI{34.97}{\mega\nothing}
\DeclareSIUnit{\pixel}{px}
\sisetup{
  detect-weight    = true,
  detect-family    = true,
  group-separator  = {\,},
  % NO ROUNDING.  round-mode=places would silently rewrite reported values
  % (0.00005 -> 0.000, 0.9989 -> 0.999).  Significant figures follow the
  % source measurement -- see thesis-formal-rules.md section 5.
  round-mode       = none,
}

% --- citations --------------------------------------------------------
% The template says the style is free choice and ships `plain`.  We use
% natbib so the prose can distinguish \citep (cite the concept) from
% \citet (name the actor) --- see thesis-style-guide.md §6.
% Swap `plainnat` -> `unsrtnat` for citation-order numbering.
\usepackage[numbers,sort&compress]{natbib}

% --- cross-references (load hyperref late) ---------------------------
\usepackage[hidelinks]{hyperref}
\usepackage[capitalise,noabbrev]{cleveref}

% --- code / identifiers ----------------------------------------------
\usepackage{listings}
\lstset{basicstyle=\ttfamily\small, breaklines=true, columns=flexible}
